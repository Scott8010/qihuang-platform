🚀 部署指南

本地开发

# 1. 激活虚拟环境
C:\Users\Administrator\.workbuddy\binaries\python\envs\default\Scripts\python.exe

# 2. 安装依赖
pip install -r requirements.txt
pip install -r requirements-test.txt  # 测试依赖

# 3. 启动服务
cd qihuang_platform
python -m uvicorn qihuang_platform.main:app --host 0.0.0.0 --port 8602

# 4. 运行测试
pytest tests/ -v
make test                     # 或使用 Makefile
python scripts/e2e_smoke.py   # E2E冒烟测试
